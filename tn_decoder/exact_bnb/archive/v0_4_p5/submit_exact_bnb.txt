#!/bin/bash
#PBS -q normal
#PBS -l select=1:ngpus=1:mpiprocs=1:ompthreads=1
#PBS -l walltime=24:00:00
#PBS -P 13004445
#PBS -j oe
#PBS -o output-bb72-p5-scan.txt
#PBS -N bb72-p5-scan

cd "${PBS_O_WORKDIR:-$(pwd)}" || exit 1

source ~/miniconda3/etc/profile.d/conda.sh
conda activate cudaqx-qec || exit 1

echo "===== ENVIRONMENT ====="
date
hostname
which python

python -c "import sys; print('Python:', sys.executable)"
python -c "import torch; print('Torch:', torch.__version__); print('CUDA available:', torch.cuda.is_available()); print('GPU count:', torch.cuda.device_count()); print('GPU:', torch.cuda.get_device_name(0) if torch.cuda.is_available() else 'NO GPU')"

echo "===== START P5 SCAN ====="
date

python scan_bb72_p5.py \
  --block-sizes 1,2,3,4,6 \
  --path-trials 6 \
  --path-samples 64 \
  --path-benchmark-repeats 2 \
  --decoder-repeats 2 \
  --output-dir p5_scan

status=$?

echo "===== FINISHED ====="
date
echo "exit status = ${status}"

exit ${status}
